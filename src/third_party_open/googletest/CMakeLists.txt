if(WIN32)
    # On Windows, build Google Test from source
    set(BUILD_GMOCK OFF)
    set(INSTALL_GTEST OFF)
    set(gtest_force_shared_crt ON)

    add_subdirectory(googletest-main)

    set_target_properties(gtest PROPERTIES FOLDER third_party_open/googletest)
    set_target_properties(gtest_main PROPERTIES FOLDER third_party_open/googletest)
else()
    # On Unix, prefer a system GTest if available; otherwise build from source.
    find_package(GTest QUIET)
    if(GTest_FOUND)
        # Try multiple variable names that find_package(GTest) might set
        if(NOT gtest_INCLUDE_DIR)
            if(DEFINED GTEST_INCLUDE_DIRS)
                set(gtest_INCLUDE_DIR ${GTEST_INCLUDE_DIRS})
            elseif(DEFINED GTest_INCLUDE_DIRS)
                set(gtest_INCLUDE_DIR ${GTest_INCLUDE_DIRS})
            elseif(TARGET GTest::gtest)
                get_target_property(gtest_INCLUDE_DIR GTest::gtest INTERFACE_INCLUDE_DIRECTORIES)
            elseif(TARGET GTest::GTest)
                get_target_property(gtest_INCLUDE_DIR GTest::GTest INTERFACE_INCLUDE_DIRECTORIES)
            endif()
        endif()
        if(NOT gtest_INCLUDE_DIR AND GTest_DIR)
            # Derive include path from GTest_DIR (e.g., <prefix>/lib64/cmake/GTest)
            get_filename_component(_gtest_prefix "${GTest_DIR}" DIRECTORY)
            get_filename_component(_gtest_prefix "${_gtest_prefix}" DIRECTORY)
            get_filename_component(_gtest_prefix "${_gtest_prefix}" DIRECTORY)
            set(gtest_INCLUDE_DIR "${_gtest_prefix}/include")
        endif()
        if(gtest_INCLUDE_DIR)
            list(GET gtest_INCLUDE_DIR 0 _gtest_include)
            set(gtest_INCLUDE_DIR "${_gtest_include}" CACHE PATH "GTest include directory" FORCE)
        endif()
        set(gtest_LIBRARY_DIR "")
        if(TARGET GTest::gtest)
            get_target_property(_gtest_location GTest::gtest IMPORTED_LOCATION_RELEASE)
            if(NOT _gtest_location)
                get_target_property(_gtest_location GTest::gtest IMPORTED_LOCATION)
            endif()
            if(_gtest_location)
                get_filename_component(_gtest_libdir "${_gtest_location}" DIRECTORY)
                set(gtest_LIBRARY_DIR "${_gtest_libdir}")
            endif()
        elseif(TARGET GTest::GTest)
            get_target_property(_gtest_location GTest::GTest IMPORTED_LOCATION_RELEASE)
            if(NOT _gtest_location)
                get_target_property(_gtest_location GTest::GTest IMPORTED_LOCATION)
            endif()
            if(_gtest_location)
                get_filename_component(_gtest_libdir "${_gtest_location}" DIRECTORY)
                set(gtest_LIBRARY_DIR "${_gtest_libdir}")
            endif()
        elseif(DEFINED GTEST_LIBRARY)
            get_filename_component(_gtest_libdir "${GTEST_LIBRARY}" DIRECTORY)
            set(gtest_LIBRARY_DIR "${_gtest_libdir}")
        endif()
        if(NOT gtest_LIBRARY_DIR AND GTest_DIR)
            get_filename_component(_gtest_libdir "${GTest_DIR}" DIRECTORY)
            get_filename_component(_gtest_libdir "${_gtest_libdir}" DIRECTORY)
            set(gtest_LIBRARY_DIR "${_gtest_libdir}")
        endif()
        if(gtest_LIBRARY_DIR)
            set(gtest_LIBRARY_DIR "${gtest_LIBRARY_DIR}" CACHE PATH "GTest library directory" FORCE)
        endif()
        if(GTest_FOUND AND NOT TARGET gtest_system)
            if(TARGET GTest::gtest)
                add_library(gtest_system ALIAS GTest::gtest)
            elseif(TARGET GTest::GTest)
                add_library(gtest_system ALIAS GTest::GTest)
            elseif(DEFINED GTEST_LIBRARY)
                add_library(gtest_system UNKNOWN IMPORTED)
                set_target_properties(gtest_system PROPERTIES IMPORTED_LOCATION "${GTEST_LIBRARY}")
                if(gtest_INCLUDE_DIR)
                    set_target_properties(gtest_system PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${gtest_INCLUDE_DIR}")
                endif()
            endif()
        endif()

        if(NOT TARGET gtest)
            if(TARGET GTest::gtest)
                add_library(gtest ALIAS GTest::gtest)
                add_library(gtest_main ALIAS GTest::gtest_main)
            elseif(TARGET GTest::GTest)
                add_library(gtest ALIAS GTest::GTest)
                if(TARGET GTest::Main)
                    add_library(gtest_main ALIAS GTest::Main)
                endif()
            elseif(DEFINED GTEST_LIBRARY)
                add_library(gtest UNKNOWN IMPORTED)
                set_target_properties(gtest PROPERTIES IMPORTED_LOCATION "${GTEST_LIBRARY}")
                if(gtest_INCLUDE_DIR)
                    set_target_properties(gtest PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${gtest_INCLUDE_DIR}")
                endif()
            endif()
        endif()
    else()
        set(BUILD_GMOCK OFF)
        set(INSTALL_GTEST OFF)
        set(gtest_force_shared_crt ON)

        add_subdirectory(googletest-main)

        set_target_properties(gtest PROPERTIES FOLDER third_party_open/googletest)
        set_target_properties(gtest_main PROPERTIES FOLDER third_party_open/googletest)
        get_target_property(gtest_INCLUDE_DIR gtest INTERFACE_INCLUDE_DIRECTORIES)
        if(gtest_INCLUDE_DIR)
            list(GET gtest_INCLUDE_DIR 0 _gtest_include)
            set(gtest_INCLUDE_DIR "${_gtest_include}" CACHE PATH "GTest include directory" FORCE)
        endif()
        get_target_property(_gtest_location gtest IMPORTED_LOCATION_RELEASE)
        if(NOT _gtest_location)
            get_target_property(_gtest_location gtest IMPORTED_LOCATION)
        endif()
        if(_gtest_location)
            get_filename_component(_gtest_libdir "${_gtest_location}" DIRECTORY)
            set(gtest_LIBRARY_DIR "${_gtest_libdir}" CACHE PATH "GTest library directory" FORCE)
        endif()
    endif()
endif()
